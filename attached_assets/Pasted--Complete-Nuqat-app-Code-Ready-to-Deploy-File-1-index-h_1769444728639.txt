javascript// State
let currentFilter = 'all';
let searchQuery = '';
let hoveredApp = null;

// DOM Elements
const dotsCanvas = document.getElementById('dots-canvas');
const appCard = document.getElementById('app-card');
const appModal = document.getElementById('app-modal');
const searchInput = document.getElementById('search-input');

// Generate semi-random but consistent dot positions
function getDotPosition(index, total) {
  const cols = Math.ceil(Math.sqrt(total));
  const col = index % cols;
  const row = Math.floor(index / cols);
  
  // Add organic randomness
  const jitterX = Math.sin(index * 1.5) * 50;
  const jitterY = Math.cos(index * 2) * 50;
  
  const left = (col / cols) * 80 + 10 + jitterX/10;
  const top = (row / Math.ceil(total/cols)) * 80 + 10 + jitterY/10;
  
  return { left: `${left}%`, top: `${top}%` };
}

// Get status badge HTML
function getStatusBadge(status) {
  const badges = {
    'live': '<span class="status-badge live">Live</span>',
    'beta': '<span class="status-badge beta">Beta</span>',
    'coming-soon': '<span class="status-badge coming-soon">Coming Soon</span>'
  };
  return badges[status] || '';
}

// Create dot elements
function renderDots() {
  dotsCanvas.innerHTML = '';
  
  APPS.forEach((app, index) => {
    const dot = document.createElement('div');
    dot.className = `dot ${app.status}`;
    dot.dataset.id = app.id;
    dot.dataset.name = app.name;
    dot.dataset.category = app.category;
    dot.style.backgroundColor = 'var(--dot-rest)';
    
    const pos = getDotPosition(index, APPS.length);
    dot.style.left = pos.left;
    dot.style.top = pos.top;
    
    // Hover events
    dot.addEventListener('mouseenter', (e) => showAppCard(app, e));
    dot.addEventListener('mousemove', (e) => moveAppCard(e));
    dot.addEventListener('mouseleave', hideAppCard);
    
    // Click to open modal
    dot.addEventListener('click', () => openModal(app));
    
    dotsCanvas.appendChild(dot);
  });
  
  applyFilters();
}

// Show floating app card
function showAppCard(app, event) {
  hoveredApp = app;
  
  // Update card content
  appCard.querySelector('.app-name').innerHTML = app.name + ' ' + getStatusBadge(app.status);
  appCard.querySelector('.app-name-arabic').textContent = app.nameArabic;
  appCard.querySelector('.app-dot').style.backgroundColor = app.color;
  appCard.querySelector('.app-description').textContent = app.description;
  appCard.querySelector('.app-platform').textContent = app.platform;
  
  // Tech tags
  const techContainer = appCard.querySelector('.app-tech');
  techContainer.innerHTML = app.tech.map(t => 
    `<span class="tech-tag">${t}</span>`
  ).join('');
  
  // Update link
  appCard.querySelector('.app-link').onclick = (e) => {
    e.preventDefault();
    openModal(app);
  };
  
  // Change dot color
  const dot = document.querySelector(`[data-id="${app.id}"]`);
  if (dot) dot.style.backgroundColor = app.color;
  
  // Show card
  appCard.classList.remove('hidden');
  moveAppCard(event);
}

// Move card with mouse
function moveAppCard(event) {
  if (!hoveredApp) return;
  
  const cardWidth = 320;
  const cardHeight = 400;
  const padding = 20;
  
  let left = event.clientX + padding;
  let top = event.clientY - 100;
  
  // Keep card within viewport
  if (left + cardWidth > window.innerWidth) {
    left = event.clientX - cardWidth - padding;
  }
  
  if (top < 0) {
    top = padding;
  }
  
  if (top + cardHeight > window.innerHeight) {
    top = window.innerHeight - cardHeight - padding;
  }
  
  appCard.style.left = `${left}px`;
  appCard.style.top = `${top}px`;
}

// Hide app card
function hideAppCard() {
  appCard.classList.add('hidden');
  
  if (hoveredApp) {
    const dot = document.querySelector(`[data-id="${hoveredApp.id}"]`);
    if (dot && !dot.classList.contains('hidden')) {
      dot.style.backgroundColor = 'var(--dot-rest)';
    }
  }
  
  hoveredApp = null;
}

// Open modal
function openModal(app) {
  hideAppCard();
  
  // Update modal content
  document.querySelector('.modal-dot').style.backgroundColor = app.color;
  document.querySelector('.modal-title').innerHTML = app.name + ' ' + getStatusBadge(app.status);
  document.querySelector('.modal-title-arabic').textContent = app.nameArabic;
  document.querySelector('.modal-platform').textContent = `${app.platform} • ${app.platformArabic}`;
  document.querySelector('.modal-description').textContent = app.description;
  
  // Tech tags
  const modalTech = document.querySelector('.modal-tech');
  modalTech.innerHTML = app.tech.map(t => 
    `<span class="tech-tag">${t}</span>`
  ).join('');
  
  // Launch button
  const launchBtn = document.querySelector('.modal-launch-btn');
  if (app.link === '#' || app.status === 'coming-soon') {
    launchBtn.textContent = 'Coming Soon';
    launchBtn.classList.add('disabled');
    launchBtn.href = '#';
  } else {
    launchBtn.textContent = 'Launch App →';
    launchBtn.classList.remove('disabled');
    launchBtn.href = app.link;
  }
  
  appModal.classList.remove('hidden');
  document.body.style.overflow = 'hidden';
}

// Close modal
function closeModal() {
  appModal.classList.add('hidden');
  document.body.style.overflow = 'auto';
}

// Filter functionality
function applyFilters() {
  const dots = document.querySelectorAll('.dot');
  
  dots.forEach(dot => {
    const app = APPS.find(a => a.id === dot.dataset.id);
    if (!app) return;
    
    const matchesFilter = currentFilter === 'all' || app.category === currentFilter;
    const matchesSearch = app.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
                         app.nameArabic.includes(searchQuery) ||
                         app.description.toLowerCase().includes(searchQuery.toLowerCase());
    
    if (matchesFilter && matchesSearch) {
      dot.classList.remove('hidden');
    } else {
      dot.classList.add('hidden');
    }
  });
  
  // Show "no results" message if all dots are hidden
  const visibleDots = document.querySelectorAll('.dot:not(.hidden)');
  const existingMessage = document.querySelector('.no-results');
  
  if (visibleDots.length === 0 && !existingMessage) {
    const message = document.createElement('div');
    message.className = 'no-results';
    message.style.cssText = 'text-align: center; padding: 4rem; color: var(--nuqat-slate); opacity: 0.6;';
    message.innerHTML = `
      <p style="font-size: 1.25rem; margin-bottom: 0.5rem;">No dots found</p>
      <p style="font-size: 0.875rem;">Try a different filter or search term</p>
    `;
    dotsCanvas.appendChild(message);
  } else if (visibleDots.length > 0 && existingMessage) {
    existingMessage.remove();
  }
}

// Event listeners
document.querySelectorAll('.filter-btn').forEach(btn => {
  btn.addEventListener('click', () => {
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    currentFilter = btn.dataset.category;
    applyFilters();
  });
});

searchInput.addEventListener('input', (e) => {
  searchQuery = e.target.value;
  applyFilters();
});

document.querySelector('.modal-close').addEventListener('click', closeModal);
appModal.addEventListener('click', (e) => {
  if (e.target === appModal) closeModal();
});

document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    closeModal();
    hideAppCard();
  }
});
