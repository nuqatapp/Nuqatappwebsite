javascript// State
let currentFilter = 'all';
let searchQuery = '';

// DOM Elements
const appGrid = document.getElementById('app-grid');
const appModal = document.getElementById('app-modal');
const searchInput = document.getElementById('search-input');
const noResults = document.getElementById('no-results');

// Get status label
function getStatusLabel(status) {
  const labels = {
    'live': 'Live',
    'beta': 'Beta',
    'coming-soon': 'Coming Soon'
  };
  return labels[status] || '';
}

// Create app card
function createAppCard(app) {
  const card = document.createElement('div');
  card.className = 'app-card';
  card.dataset.id = app.id;
  card.dataset.category = app.category;
  card.style.setProperty('--card-color', app.color);
  
  card.innerHTML = `
    <div class="card-dot" style="background-color: ${app.color};">
      ${app.platformIcon}
    </div>
    <div class="card-content">
      <h3 class="card-title">${app.name}</h3>
      <p class="card-title-arabic">${app.nameArabic}</p>
      <p class="card-description">${app.description}</p>
      <div class="card-meta">
        <span class="card-platform">${app.platformIcon} ${app.platform}</span>
        <span class="card-status ${app.status}">${getStatusLabel(app.status)}</span>
      </div>
    </div>
  `;
  
  // Click to open modal
  card.addEventListener('click', () => openModal(app));
  
  return card;
}

// Render all app cards
function renderApps() {
  appGrid.innerHTML = '';
  
  APPS.forEach((app, index) => {
    const card = createAppCard(app);
    // Stagger animation
    card.style.animationDelay = `${index * 0.1}s`;
    appGrid.appendChild(card);
  });
  
  applyFilters();
}

// Open modal with app details
function openModal(app) {
  // Update modal content
  document.querySelector('.modal-dot').style.backgroundColor = app.color;
  document.querySelector('.modal-title').textContent = app.name;
  document.querySelector('.modal-title-arabic').textContent = app.nameArabic;
  document.querySelector('.modal-platform').innerHTML = `${app.platformIcon} ${app.platform}`;
  
  const statusEl = document.querySelector('.modal-status');
  statusEl.textContent = getStatusLabel(app.status);
  statusEl.className = `modal-status ${app.status}`;
  
  document.querySelector('.modal-description').textContent = app.description;
  
  // Tech tags
  const modalTech = document.querySelector('.modal-tech');
  modalTech.innerHTML = app.tech.map(t => 
    `<span class="tech-tag">${t}</span>`
  ).join('');
  
  // Launch button
  const launchBtn = document.querySelector('.modal-launch-btn');
  if (app.link === '#' || app.status === 'coming-soon') {
    launchBtn.textContent = 'Coming Soon';
    launchBtn.classList.add('disabled');
    launchBtn.href = '#';
  } else {
    launchBtn.textContent = 'Launch App →';
    launchBtn.classList.remove('disabled');
    launchBtn.href = app.link;
  }
  
  // Show modal
  appModal.classList.remove('hidden');
  document.body.style.overflow = 'hidden';
}

// Close modal
function closeModal() {
  appModal.classList.add('hidden');
  document.body.style.overflow = 'auto';
}

// Apply filters and search
function applyFilters() {
  const cards = document.querySelectorAll('.app-card');
  let visibleCount = 0;
  
  cards.forEach(card => {
    const app = APPS.find(a => a.id === card.dataset.id);
    if (!app) return;
    
    const matchesFilter = currentFilter === 'all' || app.category === currentFilter;
    const matchesSearch = searchQuery === '' ||
      app.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
      app.nameArabic.includes(searchQuery) ||
      app.description.toLowerCase().includes(searchQuery.toLowerCase());
    
    if (matchesFilter && matchesSearch) {
      card.classList.remove('hidden');
      visibleCount++;
    } else {
      card.classList.add('hidden');
    }
  });
  
  // Show/hide no results message
  if (visibleCount === 0) {
    noResults.classList.remove('hidden');
  } else {
    noResults.classList.add('hidden');
  }
}

// Event listeners
document.querySelectorAll('.filter-btn').forEach(btn => {
  btn.addEventListener('click', () => {
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    currentFilter = btn.dataset.category;
    applyFilters();
  });
});

searchInput.addEventListener('input', (e) => {
  searchQuery = e.target.value;
  applyFilters();
});

// Modal close handlers
document.querySelector('.modal-close').addEventListener('click', closeModal);
document.querySelector('.modal-overlay').addEventListener('click', closeModal);

document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape' && !appModal.classList.contains('hidden')) {
    closeModal();
  }
});

// Prevent modal content clicks from closing modal
document.querySelector('.modal-content').addEventListener('click', (e) => {
  e.stopPropagation();
});
